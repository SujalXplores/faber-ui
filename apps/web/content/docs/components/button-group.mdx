import { Button } from "@workspace/ui/components/button";
import {
  ButtonGroup,
  ButtonGroupSeparator,
  ButtonGroupText,
} from "@workspace/ui/components/button-group";

`ButtonGroup` arranges related buttons together with shared borders, useful for toolbars and segmented actions.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add button-group
```

## Usage

```tsx
import { Button } from "@workspace/ui/components/button";
import { ButtonGroup } from "@workspace/ui/components/button-group";

export default function Example() {
  return (
    <ButtonGroup>
      <Button variant="outline">Bold</Button>
      <Button variant="outline">Italic</Button>
      <Button variant="outline">Underline</Button>
    </ButtonGroup>
  );
}
```

## API

| Component              | Description                                               |
| ---------------------- | --------------------------------------------------------- |
| `ButtonGroup`          | Root. Accepts `orientation?: "horizontal" \| "vertical"`. |
| `ButtonGroupSeparator` | Visual divider between grouped buttons.                   |
| `ButtonGroupText`      | Non-interactive text label inside the group.              |
