import { ScrollArea, ScrollBar } from "@workspace/ui/components/scroll-area";

`ScrollArea` is built on the Base UI `ScrollArea` primitive. It renders a custom-styled scrollbar while preserving native scrolling behavior.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/scroll-area.json
```

## Usage

```tsx
"use client";

import { ScrollArea } from "@workspace/ui/components/scroll-area";

export default function Example() {
  return (
    <ScrollArea className="h-48 w-72">
      <div className="p-4">
        <p>Long content here.</p>
      </div>
    </ScrollArea>
  );
}
```

## Examples

### Default

<div className="not-prose demo">
  <ScrollArea className="h-48 w-72 rounded-md border">
    <div className="p-4">
      <h4 className="mb-2 text-sm font-medium">Tags</h4>
      {Array.from({ length: 15 }).map((_, i) => (
        <div key={i} className="py-1 text-sm">Tag {i + 1}</div>
      ))}
    </div>
  </ScrollArea>
</div>

### Horizontal

<div className="not-prose demo">
