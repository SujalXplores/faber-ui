import {
  InputOTP,
  InputOTPGroup,
  InputOTPSlot,
  InputOTPSeparator,
} from "@workspace/ui/components/input-otp";

`InputOTP` is built on `input-otp`. It renders an accessible one-time password input with grouped slots and an optional separator.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/input-otp.json
```

## Usage

```tsx
"use client";

import {
  InputOTP,
  InputOTPGroup,
  InputOTPSlot,
} from "@/components/ui/input-otp";

export default function Example() {
  return (
    <InputOTP maxLength={6}>
      <InputOTPGroup>
        <InputOTPSlot index={0} />
        <InputOTPSlot index={1} />
        <InputOTPSlot index={2} />
      </InputOTPGroup>
    </InputOTP>
  );
}
```
