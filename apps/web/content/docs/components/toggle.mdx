import { Toggle } from "@workspace/ui/components/toggle";

`Toggle` is built on the Base UI `Toggle` primitive. It renders a button that switches between pressed and unpressed states.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/toggle.json
```

## Usage

```tsx
"use client";

import { Toggle } from "@/components/ui/toggle";

export default function Example() {
  return <Toggle>Bold</Toggle>;
}
```
