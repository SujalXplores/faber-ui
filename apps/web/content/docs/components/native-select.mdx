  NativeSelect,
  NativeSelectOption,
} from "@workspace/ui/components/native-select";

`NativeSelect` is a styled wrapper around the native HTML `select` element, preserving platform-specific behavior and accessibility.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add native-select
```

## Usage

```tsx
import {
  NativeSelect,
  NativeSelectOption,
} from "@workspace/ui/components/native-select";

export default function Example() {
  return (
    <NativeSelect>
      <NativeSelectOption value="apple">Apple</NativeSelectOption>
      <NativeSelectOption value="banana">Banana</NativeSelectOption>
    </NativeSelect>
  );
}
```
