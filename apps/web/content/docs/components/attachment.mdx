import {
  Attachment,

  AttachmentActions,
  AttachmentContent,
  AttachmentDescription,
  AttachmentGroup,
  AttachmentMedia,
  AttachmentTitle,
  AttachmentTrigger,
} from "@workspace/ui/components/attachment";
import { RiDownloadLine, RiFileTextLine } from "@remixicon/react";

`Attachment` is a composable card for displaying files and uploads. It supports upload states, sizes, and orientations, with optional media and action slots.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add attachment
```

## Usage

```tsx
import {
  Attachment,
  AttachmentAction,
  AttachmentActions,
  AttachmentContent,
  AttachmentDescription,
  AttachmentMedia,
  AttachmentTitle,
} from "@workspace/ui/components/attachment";
import { RiDownloadLine, RiFileTextLine } from "@remixicon/react";

export default function Example() {
  return (
    <Attachment>
      <AttachmentMedia variant="icon">
        <RiFileTextLine />
      </AttachmentMedia>
      <AttachmentContent>
        <AttachmentTitle>report.pdf</AttachmentTitle>
        <AttachmentDescription>1.2 MB</AttachmentDescription>
      </AttachmentContent>
      <AttachmentActions>
        <AttachmentAction render={<button />}>
          <RiDownloadLine />
        </AttachmentAction>
      </AttachmentActions>
    </Attachment>
  );
}
```
