import {
  Attachment,
  AttachmentAction,
  AttachmentActions,
  AttachmentContent,
  AttachmentDescription,
  AttachmentGroup,
  AttachmentMedia,
  AttachmentTitle,
  AttachmentTrigger,
} from "@workspace/ui/components/attachment";
import { RiDownloadLine, RiFileTextLine } from "@remixicon/react";

`Attachment` is a composable card for displaying files and uploads. It supports upload states, sizes, and orientations, with optional media and action slots.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add attachment
```

## Usage

```tsx
import {
  Attachment,
  AttachmentAction,
  AttachmentActions,
  AttachmentContent,
  AttachmentDescription,
  AttachmentMedia,
  AttachmentTitle,
} from "@workspace/ui/components/attachment";
import { RiDownloadLine, RiFileTextLine } from "@remixicon/react";

export default function Example() {
  return (
    <Attachment>
      <AttachmentMedia variant="icon">
        <RiFileTextLine />
      </AttachmentMedia>
      <AttachmentContent>
        <AttachmentTitle>report.pdf</AttachmentTitle>
        <AttachmentDescription>1.2 MB</AttachmentDescription>
      </AttachmentContent>
      <AttachmentActions>
        <AttachmentAction render={<button />}>
          <RiDownloadLine />
        </AttachmentAction>
      </AttachmentActions>
    </Attachment>
  );
}
```

## API

| Component               | Description                                                          |
| ----------------------- | -------------------------------------------------------------------- |
| `Attachment`            | Root card. Accepts `state`, `size`, and `orientation` props.         |
| `AttachmentGroup`       | Groups multiple attachments with shared spacing.                     |
| `AttachmentMedia`       | Icon or image slot. Accepts `variant?: "icon" \| "image"`.           |
| `AttachmentContent`     | Wraps the title and description.                                     |
| `AttachmentTitle`       | The file name or title.                                              |
| `AttachmentDescription` | Supporting text, such as file size or status.                        |
| `AttachmentActions`     | Container for action buttons.                                        |
| `AttachmentAction`      | A single action. Accepts Button props. Use `render` for composition. |
| `AttachmentTrigger`     | Trigger element for opening or uploading. Use `render`.              |
