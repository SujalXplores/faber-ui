import {
  DirectionProvider,
  useDirection,
} from "@workspace/ui/components/direction";

`Direction` is a thin re-export of the Base UI direction provider. It sets the text direction (LTR or RTL) for a subtree and exposes a `useDirection` hook to read the current value.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/direction.json
```

## Usage

```tsx
"use client";

import {
  DirectionProvider,
  useDirection,
} from "@/components/ui/direction";

export default function Example() {
  return (
    <DirectionProvider dir="rtl">
      <Content />
    </DirectionProvider>
  );
}

function Content() {
  const direction = useDirection();
  return <p>Current direction: {direction}</p>;
}
```
