} from "@workspace/ui/components/item";

`Item` is a flexible list item with slots for media, content, and actions, useful for lists, menus, and settings panels.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add item
```

## Usage

```tsx
import { Item, ItemContent, ItemDescription, ItemTitle } from "@workspace/ui/components/item";

export default function Example() {
  return (
    <Item>
      <ItemContent>
        <ItemTitle>Title</ItemTitle>
        <ItemDescription>Description</ItemDescription>
      </ItemContent>
    </Item>
  );
}
```
