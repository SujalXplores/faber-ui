import { Button } from "@workspace/ui/components/button";
import {
  Sheet,
  SheetClose,
  SheetContent,
  SheetDescription,
  SheetFooter,
  SheetHeader,
  SheetTitle,
  SheetTrigger,
} from "@workspace/ui/components/sheet";

`Sheet` is built on the Base UI `Dialog` primitive. It renders a panel that slides in from a screen edge, useful for navigation and forms.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add sheet
```

## Usage

```tsx
"use client";

import { Button } from "@workspace/ui/components/button";
import {
  Sheet,
  SheetContent,
  SheetDescription,
  SheetHeader,
  SheetTitle,
  SheetTrigger,
} from "@workspace/ui/components/sheet";

export default function Example() {
  return (
    <Sheet>
      <SheetTrigger render={<Button variant="outline">Open sheet</Button>} />
      <SheetContent>
        <SheetHeader>
          <SheetTitle>Title</SheetTitle>
          <SheetDescription>Description</SheetDescription>
        </SheetHeader>
      </SheetContent>
    </Sheet>
  );
}
```

## API

| Component          | Description                                                         |
| ------------------ | ------------------------------------------------------------------- |
| `Sheet`            | Root. Accepts `open`, `onOpenChange`, `defaultOpen`.                |
| `SheetTrigger`     | The element that opens the sheet. Use `render`.                     |
| `SheetClose`       | Closes the sheet. Use `render` to compose.                          |
| `SheetContent`     | The panel. Accepts `side?: "top" \| "right" \| "bottom" \| "left"`. |
| `SheetHeader`      | Top section with title and description.                             |
| `SheetFooter`      | Bottom section for actions.                                         |
| `SheetTitle`       | Accessible title.                                                   |
| `SheetDescription` | Muted description.                                                  |
