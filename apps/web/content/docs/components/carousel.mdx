`Carousel` is built on Embla Carousel. It supports horizontal and vertical orientations, autoplay, and snap alignment.

## Installation

```bash
pnpm dlx shadcn@latest add https://faber-ui.vercel.app/r/carousel.json
```

## Usage

```tsx
"use client";

import {
  Carousel,
  CarouselContent,
  CarouselItem,
  CarouselNext,
  CarouselPrevious,
} from "@workspace/ui/components/carousel";

export default function Example() {
  return (
    <Carousel>
      <CarouselContent>
        <CarouselItem>Item one</CarouselItem>
        <CarouselItem>Item two</CarouselItem>
      </CarouselContent>
      <CarouselPrevious />
      <CarouselNext />
    </Carousel>
  );
}
```
