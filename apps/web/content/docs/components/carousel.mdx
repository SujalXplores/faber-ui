import { Button } from "@workspace/ui/components/button";
import {
  Carousel,
  CarouselContent,
  CarouselItem,
  CarouselNext,
  CarouselPrevious,
} from "@workspace/ui/components/carousel";

`Carousel` is built on Embla Carousel. It supports horizontal and vertical orientations, autoplay, and snap alignment.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add carousel
```

## Usage

```tsx
"use client";

import {
  Carousel,
  CarouselContent,
  CarouselItem,
  CarouselNext,
  CarouselPrevious,
} from "@workspace/ui/components/carousel";

export default function Example() {
  return (
    <Carousel>
      <CarouselContent>
        <CarouselItem>Item one</CarouselItem>
        <CarouselItem>Item two</CarouselItem>
      </CarouselContent>
      <CarouselPrevious />
      <CarouselNext />
    </Carousel>
  );
}
```

## API

| Component          | Description                                              |
| ------------------ | -------------------------------------------------------- |
| `Carousel`         | Root. Accepts `orientation`, Embla options, and plugins. |
| `CarouselContent`  | The scrollable track.                                    |
| `CarouselItem`     | A single slide. Use `basis-*` to size.                   |
| `CarouselPrevious` | Previous navigation button.                              |
| `CarouselNext`     | Next navigation button.                                  |
