import {
  Message,
  MessageAvatar,
  MessageContent,
  MessageFooter,
  MessageGroup,
  MessageHeader,
} from "@workspace/ui/components/message";
import {
  Avatar,
  AvatarFallback,
  AvatarImage,
} from "@workspace/ui/components/avatar";

`Message` is a chat-style message row. It pairs an avatar with a header, content, and footer, and supports left or right alignment.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add message
```

## Usage

```tsx
import {
  Message,
  MessageAvatar,
  MessageContent,
  MessageGroup,
  MessageHeader,
} from "@workspace/ui/components/message";
import {
  Avatar,
  AvatarFallback,
} from "@workspace/ui/components/avatar";

export default function Example() {
  return (
    <MessageGroup>
      <Message>
        <MessageAvatar>
          <Avatar>
            <AvatarFallback>AB</AvatarFallback>
          </Avatar>
        </MessageAvatar>
        <MessageHeader>Anna</MessageHeader>
        <MessageContent>Hello there!</MessageContent>
      </Message>
    </MessageGroup>
  );
}
```

## API

| Component        | Description                                               |
| ---------------- | --------------------------------------------------------- |
| `MessageGroup`   | Groups a conversation thread of messages.                 |
| `Message`        | A single message row. Accepts `align?: "start" \| "end"`. |
| `MessageAvatar`  | The avatar slot.                                          |
| `MessageContent` | The message body.                                         |
| `MessageHeader`  | The sender name or header line.                           |
| `MessageFooter`  | Supporting content, such as a timestamp.                  |
