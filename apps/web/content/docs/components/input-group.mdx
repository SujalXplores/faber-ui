`InputGroup` composes an input with leading and trailing addons, such as icons, text, or buttons.

## Installation

```bash
pnpm dlx shadcn@latest add https://faber-ui.vercel.app/r/input-group.json
```

## Usage

```tsx
"use client";

import { Input } from "@workspace/ui/components/input";
import {
  InputGroup,
  InputGroupAddon,
  InputGroupInput,
} from "@workspace/ui/components/input-group";

export default function Example() {
  return (
    <InputGroup>
      <InputGroupAddon>$</InputGroupAddon>
      <InputGroupInput placeholder="0.00" />
    </InputGroup>
  );
}
```
