import { AspectRatio } from "@workspace/ui/components/aspect-ratio";

`AspectRatio` wraps content in a container that preserves a fixed aspect ratio, useful for responsive media.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/aspect-ratio.json
```

## Usage

```tsx
import { AspectRatio } from "@/components/ui/aspect-ratio";

export default function Example() {
  return (
    <AspectRatio ratio={16 / 9}>
      <img src="https://placehold.co/600x400" alt="Placeholder" className="size-full object-cover" />
    </AspectRatio>
  );
}
```
