import { ApiTable } from "@/components/api-table";

import {
  Avatar,
  AvatarBadge,
  AvatarFallback,
  AvatarGroup,
  AvatarImage,
} from "@workspace/ui/components/avatar";

`Avatar` is built on the Base UI `Avatar` primitive. It shows an image when available and falls back to initials or placeholder content while loading or on error.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add avatar
```

## Usage

```tsx
import {
  Avatar,
  AvatarFallback,
  AvatarImage,
} from "@workspace/ui/components/avatar";

export default function Example() {
  return (
    <Avatar>
      <AvatarImage src="https://github.com/shadcn.png" alt="shadcn" />
      <AvatarFallback>SC</AvatarFallback>
    </Avatar>
  );
}
```
