import {
  Tabs,
  TabsContent,
  TabsList,
  TabsTrigger,
} from "@workspace/ui/components/tabs";

`Tabs` is built on the Base UI `Tabs` primitive. It supports horizontal and vertical orientations and a `line` variant for the tab list.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/tabs.json
```

## Usage

```tsx
"use client";

import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";

export default function Example() {
  return (
    <Tabs defaultValue="tab-1">
      <TabsList>
        <TabsTrigger value="tab-1">First</TabsTrigger>
        <TabsTrigger value="tab-2">Second</TabsTrigger>
      </TabsList>
      <TabsContent value="tab-1">Content one</TabsContent>
      <TabsContent value="tab-2">Content two</TabsContent>
    </Tabs>
  );
}
```
