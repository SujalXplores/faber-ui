import { Calendar } from "@workspace/ui/components/calendar";

`Calendar` is built on React DayPicker. It renders an accessible monthly calendar for selecting single dates, ranges, or multiple dates.

## Installation

```npm
npx shadcn@latest add https://faber-ui.vercel.app/r/calendar.json
```

## Usage

```tsx
"use client";

import { useState } from "react";
import { Calendar } from "@workspace/ui/components/calendar";

export default function Example() {
  const [date, setDate] = useState<Date | undefined>(new Date());
  return <Calendar mode="single" selected={date} onSelect={setDate} />;
}
```

## Examples

### Default

<div className="not-prose demo">
