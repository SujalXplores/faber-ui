import {
  ToggleGroup,
  ToggleGroupItem,
} from "@workspace/ui/components/toggle-group";

`ToggleGroup` is built on the Base UI `ToggleGroup` primitive. It renders a set of toggle buttons that can be single or multi-select.

## Installation

```bash
cd packages/ui
pnpm dlx shadcn@latest add toggle-group
```

## Usage

```tsx
"use client";

import {
  ToggleGroup,
  ToggleGroupItem,
} from "@workspace/ui/components/toggle-group";

export default function Example() {
  return (
    <ToggleGroup type="single">
      <ToggleGroupItem value="bold">Bold</ToggleGroupItem>
      <ToggleGroupItem value="italic">Italic</ToggleGroupItem>
    </ToggleGroup>
  );
}
```

## Examples

### Single

<div className="rounded-lg border p-4">

  </ToggleGroup>

    <ToggleGroupItem value="underline">Underline</ToggleGroupItem>

## API

| Component         | Description                                                                                   |
| ----------------- | --------------------------------------------------------------------------------------------- |
| `ToggleGroup`     | Root. Accepts `type`, `value`, `defaultValue`, `onValueChange`, `orientation`, and `spacing`. |
| `ToggleGroupItem` | A single toggle. Accepts `value`.                                                             |
